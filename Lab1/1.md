# Лабораторна робота №1: Розробка та тестування HTTP-сервісів (Node.js та Go)

## 1. Мета роботи
Розробити, налаштувати та протестувати два хмарно-орієнтовані (cloud-native) HTTP-сервіси з підтримкою повного циклу CRUD (Create, Read, Update, Delete) та сховищем у пам'яті (`in-memory`). Перший сервіс реалізовано на базі **Node.js (Express)**, а другий — на базі **Go (стандартний пакет `net/http`)**.

---

## 2. Архітектура та структура проєкту
Проєкти структуровано у вигляді двох ізольованих мікросервісів у загальній робочій директорії `cloud-native-lab1/`:
* `cloud-native-lab1/node-service/` — Node.js додаток на фреймворку Express (порт `3000`).
* `cloud-native-lab1/go-service/` — Go додаток з використанням вбудованого маршрутизатора `net/http` та потокобезпечного м'ютекса `sync.Mutex` (порт `8080`).

Обидва сервіси реалізують ідентичний REST API контракт:
1. `GET /health` — перевірка працездатності та стану сервісу.
2. `GET /items` — отримання списку всіх товарів.
3. `POST /items` — створення нового товару.
4. `GET /items/{id}` — отримання конкретного товару за ідентифікатором.
5. `PUT /items/{id}` — оновлення даних існуючого товару.
6. `DELETE /items/{id}` — видалення товару за ідентифікатором.

---

## 3. Реалізація Node.js сервісу (Express)
Сервіс працює на порту `3000` та використовує вбудований middleware `express.json()` для парсингу JSON-тіла запитів.

### Код сервера (`node-service/server.js`):
```javascript
const express = require('express');
const app = express();
const PORT = process.env.PORT || 3000;

app.use(express.json());

let items = {
    1: { id: 1, name: 'Laptop', price: 1200 },
    2: { id: 2, name: 'Mechanical Keyboard', price: 150 }
};
let nextId = 3;

// Health check
app.get('/health', (req, res) => {
    res.status(200).json({ status: 'OK', service: 'node-service', timestamp: new Date().toISOString() });
});

// GET all & POST
app.get('/items', (req, res) => {
    res.json(Object.values(items));
});

app.post('/items', (req, res) => {
    const { name, price } = req.body;
    if (!name || typeof price !== 'number') {
        return res.status(400).json({ error: 'Name and valid price are required' });
    }
    const newItem = { id: nextId++, name, price };
    items[newItem.id] = newItem;
    res.status(201).json(newItem);
});

// GET by ID, PUT, DELETE
app.get('/items/:id', (req, res) => {
    const item = items[req.params.id];
    if (!item) return res.status(404).json({ error: 'Item not found' });
    res.json(item);
});

app.put('/items/:id', (req, res) => {
    const item = items[req.params.id];
    if (!item) return res.status(404).json({ error: 'Item not found' });
    const { name, price } = req.body;
    if (name) item.name = name;
    if (price) item.price = price;
    res.json(item);
});

app.delete('/items/:id', (req, res) => {
    if (!items[req.params.id]) return res.status(404).json({ error: 'Item not found' });
    delete items[req.params.id];
    res.status(204).send();
});

app.listen(PORT, () => {
    console.log(`Node.js service is running on http://localhost:${PORT}`);
});
```

## 4. Реалізація Go сервісу (`net/http`)
Сервіс працює на порту `8080`, написаний виключно засобами стандартної бібліотеки Go без сторонніх фреймворків і забезпечує потокобезпеку (Concurrency Safety) при одночасних HTTP-запитах завдяки використанню м'ютекса `sync.Mutex`.

### Код сервера (`go-service/main.go`):
```go
package main

import (
	"encoding/json"
	"net/http"
	"strconv"
	"strings"
	"sync"
	"time"
)

type Item struct {
	ID    int     `json:"id"`
	Name  string  `json:"name"`
	Price float64 `json:"price"`
}

type MemoryStore struct {
	mu     sync.Mutex
	items  map[int]Item
	nextID int
}

var store = MemoryStore{
	items: map[int]Item{
		1: {ID: 1, Name: "Laptop", Price: 1200},
		2: {ID: 2, Name: "Mechanical Keyboard", Price: 150},
	},
	nextID: 3,
}

func main() {
	mux := http.NewServeMux()
	mux.HandleFunc("/health", healthCheckHandler)
	mux.HandleFunc("/items", itemsHandler)
	mux.HandleFunc("/items/", itemDetailHandler)

	server := &http.Server{
		Addr:         ":8080",
		Handler:      mux,
		ReadTimeout:  10 * time.Second,
		WriteTimeout: 10 * time.Second,
	}

	println("Go service is running on http://localhost:8080")
	if err := server.ListenAndServe(); err != nil {
		println("Server failed:", err.Error())
	}
}

func healthCheckHandler(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
		return
	}
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(http.StatusOK)
	json.NewEncoder(w).Encode(map[string]interface{}{
		"status":    "OK",
		"service":   "go-service",
		"timestamp": time.Now().Format(time.RFC3339),
	})
}

func itemsHandler(w http.ResponseWriter, r *http.Request) {
	switch r.Method {
	case http.MethodGet:
		store.mu.Lock()
		itemList := make([]Item, 0, len(store.items))
		for _, item := range store.items {
			itemList = append(itemList, item)
		}
		store.mu.Unlock()

		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(http.StatusOK)
		json.NewEncoder(w).Encode(itemList)

	case http.MethodPost:
		var newItem Item
		if err := json.NewDecoder(r.Body).Decode(&newItem); err != nil {
			http.Error(w, err.Error(), http.StatusBadRequest)
			return
		}
		if newItem.Name == "" || newItem.Price <= 0 {
			http.Error(w, "Name and valid price are required", http.StatusBadRequest)
			return
		}

		store.mu.Lock()
		newItem.ID = store.nextID
		store.nextID++
		store.items[newItem.ID] = newItem
		store.mu.Unlock()

		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(http.StatusCreated)
		json.NewEncoder(w).Encode(newItem)

	default:
		http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
	}
}

func itemDetailHandler(w http.ResponseWriter, r *http.Request) {
	parts := strings.Split(strings.TrimPrefix(r.URL.Path, "/items/"), "/")
	if len(parts) == 0 || parts[0] == "" {
		http.NotFound(w, r)
		return
	}

	id, err := strconv.Atoi(parts[0])
	if err != nil {
		http.Error(w, "Invalid ID", http.StatusBadRequest)
		return
	}

	store.mu.Lock()
	defer store.mu.Unlock()

	item, exists := store.items[id]

	switch r.Method {
	case http.MethodGet:
		if !exists {
			http.Error(w, `{"error":"Item not found"}`, http.StatusNotFound)
			return
		}
		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(http.StatusOK)
		json.NewEncoder(w).Encode(item)

	case http.MethodPut:
		if !exists {
			http.Error(w, `{"error":"Item not found"}`, http.StatusNotFound)
			return
		}
		var updatedItem Item
		if err := json.NewDecoder(r.Body).Decode(&updatedItem); err != nil {
			http.Error(w, err.Error(), http.StatusBadRequest)
			return
		}
		if updatedItem.Name != "" {
			item.Name = updatedItem.Name
		}
		if updatedItem.Price > 0 {
			item.Price = updatedItem.Price
		}
		store.items[id] = item

		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(http.StatusOK)
		json.NewEncoder(w).Encode(item)

	case http.MethodDelete:
		if !exists {
			http.Error(w, `{"error":"Item not found"}`, http.StatusNotFound)
			return
		}
		delete(store.items, id)
		w.WriteHeader(http.StatusNoContent)

	default:
		http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
	}
}
```

## 5. Результати тестування

### 5.1. Тестування Node.js сервісу (`http://localhost:3000`)
Для коректної передачі JSON-даних у середовищі Windows PowerShell використано командлет `Invoke-RestMethod`:

* **`POST /items`** (створення товару `{"name":"Gaming Mouse","price":80}`):
  ```text
  id name         price
  -- ----         -----
   3 Gaming Mouse    80
  ```
* **`GET /items/3`** (отримання за ID):
  ```text
  id name         price
  -- ----         -----
   3 Gaming Mouse    80
  ```
* **`PUT /items/3`** (оновлення товару `{"name":"Wireless Gaming Mouse","price":95}`):
  ```text
  id name                  price
  -- ----                  -----
   3 Wireless Gaming Mouse    95
  ```
* **`DELETE /items/3`** — виконано успішно (статус `204 No Content`).

### 5.2. Тестування Go сервісу (`http://localhost:8080`)
* **`POST /items`** (створення товару `{"name":"Go Keyboard","price":120}`):
  ```text

  -- ----        -----
   3 Go Keyboard   120
  ```
* **`GET /items/3`** (отримання за ID):

## 6. Висновки
У ході виконання лабораторної роботи було успішно спроєктовано, розроблено та протестовано два RESTful HTTP-сервіси на платформах **Node.js (Express)** та **Go (`net/http`)**. 

Під час роботи:
* Реалізовано повний цикл CRUD-операцій з використанням `in-memory` сховища даних та коректними HTTP-статусами (`200 OK`, `201 Created`, `204 No Content`, `400 Bad Request`, `404 Not Found`).
* Забезпечено потокобезпеку при роботі зі спільною пам'яттю в Go за допомогою `sync.Mutex`.
* Вирішено проблему екранування лапок при передачі JSON-запитів у Windows PowerShell шляхом переходу з зовнішньої утиліти `curl.exe` на вбудований командлет `Invoke-RestMethod`.
